1.使用传统基于所有维度的预聚合 vs 基于预测的查询维度预聚合
1m06s,1m29s vs 52s,46s
```
SELECT
  TIME_FLOOR(__time, 'PT1H') AS "hour",  
  "appsysid",
  "appid",               
  "group",
  "status_code",
  SUM("dur_max") AS "dur_max",               
  SUM("dur_sum") AS "dur_sum"                
FROM dwm_request
WHERE
  __time BETWEEN TIMESTAMP '2019-01-11 09:00:00' 
    AND TIMESTAMP '2025-02-16 10:00:00'
GROUP BY 1, 2, 3, 4, 5            
ORDER BY "hour" ASC   
limit 1000
```
